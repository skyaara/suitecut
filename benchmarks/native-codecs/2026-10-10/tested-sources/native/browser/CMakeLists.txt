cmake_minimum_required(VERSION 3.21)
project(suitecut_browser LANGUAGES C CXX)
if(APPLE)
  enable_language(OBJCXX)
endif()
if(WIN32)
  message(FATAL_ERROR "Windows packaging and sandbox support are not yet implemented. Do not ship an unsandboxed substitute.")
endif()
if(NOT IS_ABSOLUTE "${CEF_ROOT}")
  message(FATAL_ERROR "Pass -DCEF_ROOT=/absolute/path/to/a/CEF/binary/distribution")
endif()
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
set(USE_SANDBOX ON CACHE BOOL "Keep Chromium renderer sandbox enabled" FORCE)
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
if(NOT IS_ABSOLUTE "${LIBYUV_ROOT}")
  message(FATAL_ERROR "Pass -DLIBYUV_ROOT=/absolute/path/to/the/pinned/libyuv/source")
endif()
set(LIBYUV_DISABLE_JPEG ON CACHE BOOL "Only raw pixel conversion is needed" FORCE)
add_subdirectory("${LIBYUV_ROOT}" libyuv EXCLUDE_FROM_ALL)
set(CMAKE_CXX_STANDARD 20)
enable_testing()
set(SUITECUT_FFMPEG_ROOT "" CACHE PATH "Optional FFmpeg development sysroot")
find_path(AV_INCLUDE libavcodec/avcodec.h HINTS "${SUITECUT_FFMPEG_ROOT}/usr/include/x86_64-linux-gnu" "${SUITECUT_FFMPEG_ROOT}/include" REQUIRED)
foreach(component avcodec avformat avutil swscale swresample avfilter)
  find_library(${component}_LIBRARY NAMES ${component} HINTS "${SUITECUT_FFMPEG_ROOT}/usr/lib/x86_64-linux-gnu" "${SUITECUT_FFMPEG_ROOT}/lib" REQUIRED)
endforeach()
add_library(suitecut-media STATIC media_engine.cc media_file.cc media_render.cc)
target_include_directories(suitecut-media PRIVATE "${AV_INCLUDE}")
target_link_libraries(suitecut-media PUBLIC ${avformat_LIBRARY} ${avcodec_LIBRARY} ${avutil_LIBRARY} ${swscale_LIBRARY} ${swresample_LIBRARY} ${avfilter_LIBRARY})
add_executable(suitecut-media-test media_engine_test.cc)
target_link_libraries(suitecut-media-test suitecut-media)
add_test(NAME native-media COMMAND suitecut-media-test)
add_executable(suitecut-codec-test media_codec_test.cc)
target_link_libraries(suitecut-codec-test suitecut-media)
add_test(NAME native-codecs COMMAND suitecut-codec-test)
add_executable(suitecut-audio-clock-test audio_clock_test.cc)
add_test(NAME native-audio-clock COMMAND suitecut-audio-clock-test)
add_executable(suitecut-pixel-test pixel_test.cc)
target_include_directories(suitecut-pixel-test PRIVATE "${LIBYUV_ROOT}/include")
target_link_libraries(suitecut-pixel-test yuv)
add_test(NAME native-pixel-conversion COMMAND suitecut-pixel-test)

if(APPLE)
  add_executable(suitecut-browser MACOSX_BUNDLE browser.cc main_mac.mm)
  set_target_properties(suitecut-browser PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in")
  SET_EXECUTABLE_TARGET_PROPERTIES(suitecut-browser)
  target_link_libraries(suitecut-browser libcef_dll_wrapper ${CEF_STANDARD_LIBS})
  # CEF requires a helper app for each subprocess role, including the renderer sandbox.
  foreach(entry ${CEF_HELPER_APP_SUFFIXES})
    string(REPLACE ":" ";" parts "${entry}")
    list(GET parts 0 name_suffix)
    list(GET parts 1 target_suffix)
    set(helper "suitecut-helper${target_suffix}")
    set(helper_name "suitecut-browser Helper${name_suffix}")
    add_executable(${helper} MACOSX_BUNDLE helper_mac.cc)
    SET_EXECUTABLE_TARGET_PROPERTIES(${helper})
    target_link_libraries(${helper} libcef_dll_wrapper ${CEF_STANDARD_LIBS})
    set_target_properties(${helper} PROPERTIES OUTPUT_NAME "${helper_name}" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/suitecut-browser.app/Contents/Frameworks")
    add_dependencies(suitecut-browser ${helper})
  endforeach()
  add_custom_command(TARGET suitecut-browser POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CEF_ROOT}/Release/Chromium Embedded Framework.framework" "$<TARGET_BUNDLE_DIR:suitecut-browser>/Contents/Frameworks/Chromium Embedded Framework.framework"
    COMMAND ${CMAKE_COMMAND} -E copy "${CEF_ROOT}/LICENSE.txt" "$<TARGET_BUNDLE_DIR:suitecut-browser>/Contents/Resources/CEF-LICENSE.txt"
    COMMAND ${CMAKE_COMMAND} -E copy "${CEF_ROOT}/CREDITS.html" "$<TARGET_BUNDLE_DIR:suitecut-browser>/Contents/Resources/CEF-CREDITS.html")
else()
  add_executable(suitecut-browser browser.cc main_linux.cc)
  SET_EXECUTABLE_TARGET_PROPERTIES(suitecut-browser)
  ADD_LOGICAL_TARGET("libcef_lib" "${CEF_LIB_DEBUG}" "${CEF_LIB_RELEASE}")
  target_link_libraries(suitecut-browser libcef_lib libcef_dll_wrapper ${CEF_STANDARD_LIBS})
  set_target_properties(suitecut-browser PROPERTIES BUILD_RPATH "$ORIGIN")
  add_custom_command(TARGET suitecut-browser POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CEF_ROOT}/Release" "$<TARGET_FILE_DIR:suitecut-browser>"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CEF_ROOT}/Resources" "$<TARGET_FILE_DIR:suitecut-browser>"
    COMMAND ${CMAKE_COMMAND} -E copy "${CEF_ROOT}/LICENSE.txt" "$<TARGET_FILE_DIR:suitecut-browser>/CEF-LICENSE.txt"
    COMMAND ${CMAKE_COMMAND} -E copy "${CEF_ROOT}/CREDITS.html" "$<TARGET_FILE_DIR:suitecut-browser>/CEF-CREDITS.html")
endif()
target_include_directories(suitecut-browser PRIVATE "${CEF_ROOT}")
target_include_directories(suitecut-browser PRIVATE "${LIBYUV_ROOT}/include")
target_link_libraries(suitecut-browser yuv)
target_link_libraries(suitecut-browser suitecut-media)
# CEF defaults to -fno-exceptions; native media errors cross this controlled boundary.
target_compile_options(suitecut-browser PRIVATE -fexceptions)
if(APPLE)
  add_custom_command(TARGET suitecut-browser POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy "${LIBYUV_ROOT}/LICENSE" "$<TARGET_BUNDLE_DIR:suitecut-browser>/Contents/Resources/libyuv-LICENSE.txt")
else()
  add_custom_command(TARGET suitecut-browser POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy "${LIBYUV_ROOT}/LICENSE" "$<TARGET_FILE_DIR:suitecut-browser>/libyuv-LICENSE.txt")
endif()
