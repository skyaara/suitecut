cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(suitecut_native_proof_tools LANGUAGES CXX)
  enable_testing()
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(Threads REQUIRED)
  add_executable(suitecut-proof-supervisor proof_supervisor.cc)
  add_executable(suitecut-proof-supervisor-fixture proof_supervisor_fixture.cc)
  add_executable(suitecut-proof-supervisor-observer-test proof_supervisor_observer_test.cc)
  foreach(target IN ITEMS suitecut-proof-supervisor suitecut-proof-supervisor-fixture suitecut-proof-supervisor-observer-test)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../runner")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -fexceptions)
  endforeach()
  target_compile_options(suitecut-proof-supervisor-fixture PRIVATE -UNDEBUG)
  target_compile_options(suitecut-proof-supervisor-observer-test PRIVATE -UNDEBUG)
  target_link_libraries(suitecut-proof-supervisor PRIVATE Threads::Threads)
  target_link_libraries(suitecut-proof-supervisor-observer-test PRIVATE Threads::Threads)
  add_test(NAME native-proof-supervisor-observer-ownership COMMAND suitecut-proof-supervisor-observer-test)
  find_program(SUITECUT_PROOF_PYTHON NAMES python3)
  if(SUITECUT_PROOF_PYTHON)
    add_test(NAME native-proof-supervisor-processes COMMAND "${SUITECUT_PROOF_PYTHON}"
      "${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/check-native-proof-supervisor.py"
      --executable "$<TARGET_FILE:suitecut-proof-supervisor>"
      --fixture "$<TARGET_FILE:suitecut-proof-supervisor-fixture>"
      --directory "${CMAKE_CURRENT_BINARY_DIR}/proof-supervisor-data")
    set_tests_properties(native-proof-supervisor-processes PROPERTIES TIMEOUT 30)
  endif()
  set_tests_properties(native-proof-supervisor-observer-ownership PROPERTIES TIMEOUT 10)
endif()
