cmake_minimum_required(VERSION 3.20)
project(SuiteCutResourcePolicy LANGUAGES CXX)
include(CTest)
find_package(Threads REQUIRED)

# No CEF, browser runtime, or platform SDK transport is needed for these checks.
foreach(name IN ITEMS worker_budget trace_background)
  add_executable(${name}_test ../browser/${name}_test.cc)
  target_compile_features(${name}_test PRIVATE cxx_std_20)
  target_link_libraries(${name}_test PRIVATE Threads::Threads)
  if(MSVC)
    target_compile_options(${name}_test PRIVATE /UNDEBUG /W4)
  else()
    target_compile_options(${name}_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
  endif()
  add_test(NAME ${name} COMMAND ${name}_test)
  set_tests_properties(${name} PROPERTIES TIMEOUT 15)
endforeach()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(thread_start_failure_test ../browser/trace_background_start_failure_test.cc)
  target_compile_features(thread_start_failure_test PRIVATE cxx_std_20)
  target_compile_options(thread_start_failure_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
  target_link_libraries(thread_start_failure_test PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
  add_test(NAME thread_start_failure COMMAND thread_start_failure_test)
  set_tests_properties(thread_start_failure PROPERTIES TIMEOUT 15)
endif()
