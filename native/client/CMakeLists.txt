cmake_minimum_required(VERSION 3.20)
project(SuiteCutClient VERSION 0.1.0 LANGUAGES C CXX)

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(suitecut_standalone OFF)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  set(suitecut_standalone ON)
  include(CTest)
endif()
set(suitecut_tests_default OFF)
if(suitecut_standalone)
  set(suitecut_tests_default ${BUILD_TESTING})
endif()
option(SUITECUT_CLIENT_BUILD_TESTS "Build SDK unit and integration executables" ${suitecut_tests_default})
option(SUITECUT_CLIENT_BUILD_EXAMPLES "Build SDK command-line examples" ${suitecut_standalone})
option(SUITECUT_CLIENT_SANITIZERS "Enable address and undefined behavior checks" OFF)

find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(ZLIB REQUIRED)
include(cmake/Modules.cmake)
add_subdirectory(src)

# Object modules enforce private include boundaries and compile once. Keep one
# archive/dylib and the existing C ABI instead of introducing inter-library ABI
# and static archive ordering requirements for existing SDK consumers.
add_library(suitecut_client)
add_library(SuiteCut::Client ALIAS suitecut_client)
foreach(module IN LISTS suitecut_client_modules)
  target_sources(suitecut_client PRIVATE $<TARGET_OBJECTS:${module}>)
endforeach()
set_target_properties(suitecut_client PROPERTIES
  EXPORT_NAME Client VERSION ${PROJECT_VERSION} SOVERSION 0)
target_compile_features(suitecut_client PUBLIC cxx_std_20)
target_include_directories(suitecut_client PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(suitecut_client PRIVATE Threads::Threads OpenSSL::SSL OpenSSL::Crypto ZLIB::ZLIB)
if(WIN32)
  target_link_libraries(suitecut_client PRIVATE ws2_32)
  if(BUILD_SHARED_LIBS)
    target_compile_definitions(suitecut_sdk_c_api PRIVATE SUITECUT_CLIENT_SHARED SUITECUT_CLIENT_BUILD)
    target_compile_definitions(suitecut_client INTERFACE SUITECUT_CLIENT_SHARED)
  endif()
endif()
if(SUITECUT_CLIENT_SANITIZERS)
  target_compile_options(suitecut_client INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(suitecut_client PUBLIC -fsanitize=address,undefined)
endif()

if(SUITECUT_CLIENT_BUILD_EXAMPLES)
  add_subdirectory(examples)
endif()
if(SUITECUT_CLIENT_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
include(cmake/Install.cmake)
