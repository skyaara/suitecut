cmake_minimum_required(VERSION 3.20)
project(SuiteCutClient VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(ZLIB REQUIRED)
option(SUITECUT_CLIENT_SANITIZERS "Enable address, undefined behavior and leak checks" OFF)
option(SUITECUT_CLIENT_BUILD_EXAMPLES "Build native control example" ON)

add_library(suitecut_client src/client.cpp src/transport.cpp src/json.cpp src/decoder.cpp src/browser.cpp
  src/observation.cpp src/agent.cpp src/agent_driver.cpp src/simple_agent.cpp src/agent_tools.cpp src/visual.cpp
  src/cdp_backend.cpp src/cdp_connection.cpp src/cdp_dom.cpp src/cdp_input.cpp src/cdp_page.cpp
  src/cdp_frame.cpp src/cdp_network.cpp
  src/bidi_backend.cpp src/bidi_dom.cpp src/bidi_features.cpp src/bidi_har.cpp src/bidi_privileged.cpp src/bidi_permissions.cpp src/credentials.cpp src/image.cpp src/cdp_media.cpp)
add_library(SuiteCut::Client ALIAS suitecut_client)
set_target_properties(suitecut_client PROPERTIES EXPORT_NAME Client VERSION ${PROJECT_VERSION} SOVERSION 0)
target_compile_features(suitecut_client PUBLIC cxx_std_20)
target_include_directories(suitecut_client PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(suitecut_client PUBLIC Threads::Threads OpenSSL::SSL OpenSSL::Crypto ZLIB::ZLIB)
target_compile_options(suitecut_client PRIVATE -Wall -Wextra -Wpedantic)
if(WIN32)
  target_link_libraries(suitecut_client PRIVATE ws2_32)
endif()
if(SUITECUT_CLIENT_SANITIZERS)
  target_compile_options(suitecut_client PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(suitecut_client PUBLIC -fsanitize=address,undefined)
endif()
if(SUITECUT_CLIENT_BUILD_EXAMPLES)
  add_executable(suitecut-native-control examples/control.cpp examples/control_cli.cpp examples/control_session.cpp)
  target_link_libraries(suitecut-native-control PRIVATE SuiteCut::Client)
  add_executable(suitecut-native-record examples/record.cpp)
  target_link_libraries(suitecut-native-record PRIVATE SuiteCut::Client)
  if(SUITECUT_CLIENT_SANITIZERS)
    target_compile_definitions(suitecut-native-control PRIVATE SUITECUT_CLI_SANITIZERS=1)
  endif()
  add_executable(suitecut-native-chromium examples/chromium.cpp)
  target_link_libraries(suitecut-native-chromium PRIVATE SuiteCut::Client)
  add_executable(suitecut-native-agent examples/agent.cpp)
  target_link_libraries(suitecut-native-agent PRIVATE SuiteCut::Client)
  add_executable(suitecut-native-command examples/raw.c)
  target_link_libraries(suitecut-native-command PRIVATE SuiteCut::Client)
  add_executable(suitecut-native-read-window examples/read_window.cpp)
  target_link_libraries(suitecut-native-read-window PRIVATE SuiteCut::Client)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set_target_properties(suitecut-native-agent suitecut-native-command suitecut-native-read-window suitecut-native-chromium suitecut-native-control suitecut-native-record PROPERTIES
      INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
  endif()
  install(TARGETS suitecut-native-agent suitecut-native-command suitecut-native-read-window suitecut-native-chromium suitecut-native-control suitecut-native-record
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()

if(BUILD_TESTING)
  # Test sources are intentionally local to the standalone client: configuring
  # this project never discovers a CEF SDK or builds Chromium/the engine.
  file(GLOB CLIENT_TEST_SOURCES CONFIGURE_DEPENDS tests/*_test.cpp)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/transport_fake_engine.cpp")
    add_executable(transport_fake_engine tests/transport_fake_engine.cpp)
    target_link_libraries(transport_fake_engine PRIVATE SuiteCut::Client)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/browser_facade_engine.cpp")
    add_executable(browser_facade_engine tests/browser_facade_engine.cpp)
    target_link_libraries(browser_facade_engine PRIVATE SuiteCut::Client)
  endif()
  foreach(source IN LISTS CLIENT_TEST_SOURCES)
    get_filename_component(name "${source}" NAME_WE)
    add_executable(${name} "${source}")
    target_include_directories(${name} PRIVATE src)
    target_link_libraries(${name} PRIVATE SuiteCut::Client)
    if(name STREQUAL "control_cli_test")
      target_sources(${name} PRIVATE examples/control_cli.cpp)
      target_include_directories(${name} PRIVATE examples)
    endif()
    if(name STREQUAL "browser_facade_test")
      add_test(NAME ${name} COMMAND ${name} $<TARGET_FILE:browser_facade_engine>)
    else()
      add_test(NAME ${name} COMMAND ${name})
    endif()
    set_tests_properties(${name} PROPERTIES TIMEOUT 60)
  endforeach()
  add_executable(c_abi_test tests/c_abi_test.c)
  target_link_libraries(c_abi_test PRIVATE SuiteCut::Client)
  add_test(NAME c_abi_test COMMAND c_abi_test)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/chromium_real_browser.cpp")
    add_executable(suitecut-native-chromium-real-browser tests/chromium_real_browser.cpp)
    target_link_libraries(suitecut-native-chromium-real-browser PRIVATE SuiteCut::Client)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/chromium_workflows.cpp")
    add_executable(suitecut-native-chromium-workflows tests/chromium_workflows.cpp)
    target_link_libraries(suitecut-native-chromium-workflows PRIVATE SuiteCut::Client)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/chromium_attach.cpp")
    add_executable(suitecut-native-chromium-attach tests/chromium_attach.cpp)
    target_link_libraries(suitecut-native-chromium-attach PRIVATE SuiteCut::Client)
    target_include_directories(suitecut-native-chromium-attach PRIVATE src)
  endif()
  foreach(workflow IN ITEMS gap_agent_workflows firefox_workflows firefox_attach media_workflows tls_workflows cross_browser_features firefox_drag_workflows)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/${workflow}.cpp")
      add_executable(suitecut-native-${workflow} tests/${workflow}.cpp)
      target_include_directories(suitecut-native-${workflow} PRIVATE src)
      target_link_libraries(suitecut-native-${workflow} PRIVATE SuiteCut::Client)
    endif()
  endforeach()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/real_browser.cpp")
    add_executable(native-client-real-browser tests/real_browser.cpp)
    target_link_libraries(native-client-real-browser PRIVATE SuiteCut::Client)
  endif()
endif()

install(TARGETS suitecut_client EXPORT SuiteCutClientTargets
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/suitecut DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES README.md VALIDATION.md ../../LICENSE DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/doc/SuiteCutClient)
install(DIRECTORY skills/ DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/SuiteCutClient/skills)
install(EXPORT SuiteCutClientTargets NAMESPACE SuiteCut:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SuiteCutClient)
configure_package_config_file(SuiteCutClientConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/SuiteCutClientConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SuiteCutClient)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/SuiteCutClientConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/SuiteCutClientConfig.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/SuiteCutClientConfigVersion.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SuiteCutClient)
