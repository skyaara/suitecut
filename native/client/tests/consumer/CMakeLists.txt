cmake_minimum_required(VERSION 3.20)
project(SuiteCutInstalledConsumer LANGUAGES C)
option(SUITECUT_CONSUMER_C_ONLY "Verify a shared SDK from a C-only project" OFF)
if(NOT SUITECUT_CONSUMER_C_ONLY)
  enable_language(CXX)
endif()
if(SUITECUT_CONSUMER_SOURCE)
  add_subdirectory("${SUITECUT_CONSUMER_SOURCE}" sdk)
else()
  find_package(SuiteCutClient 0.1 CONFIG REQUIRED)
endif()
enable_testing()
add_executable(c_consumer c_consumer.c)
target_link_libraries(c_consumer PRIVATE SuiteCut::Client)
add_test(NAME installed_c_abi COMMAND c_consumer)
if(NOT SUITECUT_CONSUMER_C_ONLY)
  add_executable(cpp_consumer cpp_consumer.cpp)
  target_link_libraries(cpp_consumer PRIVATE SuiteCut::Client)
  add_test(NAME installed_cpp_api COMMAND cpp_consumer)
  # Compile each public header first and alone, with no umbrella or private includes.
  foreach(header IN ITEMS client.h browser.hpp transport.hpp json.hpp credentials.hpp image.hpp
      observation.hpp visual.hpp agent.hpp agent_driver.hpp agent_tools.hpp simple_agent.hpp)
    string(REPLACE "." "_" name "${header}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${name}.cpp" "#include <suitecut/${header}>\n")
    add_library(header_${name} OBJECT "${CMAKE_CURRENT_BINARY_DIR}/${name}.cpp")
    target_link_libraries(header_${name} PRIVATE SuiteCut::Client)
  endforeach()
endif()
set_tests_properties(installed_c_abi PROPERTIES TIMEOUT 10)
if(TEST installed_cpp_api)
  set_tests_properties(installed_cpp_api PROPERTIES TIMEOUT 10)
endif()
